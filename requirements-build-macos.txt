# KVRouite - additional packages needed to BUILD the macOS bundle.
#
#     pip install -r requirements.txt
#     pip install -r requirements-build-macos.txt
#     python3 build_macos.py
#
# Why a separate file and not requirements-build.txt: that one carries pefile
# and pywin32-ctypes, which exist for the Windows executable only. They would
# install on a Mac and sit there unused. The versions below are the same ones
# the Windows build is pinned to, so both platforms are built with the same
# PyInstaller.

altgraph==0.17.5
packaging==26.3
pyinstaller==6.22.2
pyinstaller-hooks-contrib==2026.7
setuptools==84.0.0
